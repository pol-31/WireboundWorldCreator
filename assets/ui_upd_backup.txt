float gridSize = 1.0f;
for (float t = 0.0f; t < maxDistance; t += gridSize * 0.5f) {
    glm::vec3 p = rayOrigin + rayDir * t;
    int x = floor(p.x + halfMapSize);
    int z = floor(p.z + halfMapSize);

    if (x < 0 || x >= mapWidth || z < 0 || z >= mapHeight) continue;

    float height = heightmap[z * mapWidth + x]; // read CPU-side heightmap
    if (p.y <= height) {
        // Hit terrain
        return glm::vec3(p.x, height, p.z);
    }
